function dp2px(dp)
  return math.floor(activity.getResources().getDisplayMetrics().density * dp + 0.5)
end

local M3 = {
  Background     = 0xFFFEF7FF,
  Primary        = 0xFF6750A4,
  OnPrimary      = 0xFFFFFFFF,
  SecondaryContainer = 0xFFE8DEF8,
  OnSecondaryContainer = 0xFF1D192B,
  SurfaceVariant = 0xFFE7E0EC,
  OnSurface      = 0xFF1D1B20,
  OnSurfaceSub   = 0xFF49454F,
  Outline        = 0xFF79747E,
}

    M3.SecondaryContainer = res.getColor(android.R.color.system_accent2_100)
    M3.SurfaceVariant = res.getColor(android.R.color.system_neutral2_100)
  end)
end

function createM3Shape(color, radiusDp)
  local drawable = GradientDrawable()
  drawable.setColor(color)
  drawable.setCornerRadius(dp2px(radiusDp))
  return drawable
end

function createM3OutlinedShape(strokeColor, radiusDp)
  local drawable = GradientDrawable()
  drawable.setCornerRadius(dp2px(radiusDp))
  drawable.setStroke(dp2px(1), strokeColor)
  return drawable
end

function createMaterialSwitch(defaultChecked, onCheckChanged)
  local isChecked = defaultChecked or false
  local trackW = dp2px(52)
  local trackH = dp2px(32)
  local thumbOffSize = dp2px(16)
  local thumbOnSize = dp2px(24)
  local marginMin = dp2px(8)
  local marginMax = dp2px(4)

  local trackDrawable = GradientDrawable()
  trackDrawable.setCornerRadius(trackH / 2)

  local thumbDrawable = GradientDrawable()
  thumbDrawable.setCornerRadius(thumbOnSize / 2)

  local switchFrame = FrameLayout(activity)
  switchFrame.setLayoutParams(ViewGroup.LayoutParams(trackW, trackH))

  local trackView = View(activity)
  trackView.setLayoutParams(FrameLayout.LayoutParams(-1, -1))
  trackView.setBackground(trackDrawable)
  switchFrame.addView(trackView)

  local thumbView = View(activity)
  local thumbParams = FrameLayout.LayoutParams(thumbOffSize, thumbOffSize)
  thumbParams.gravity = Gravity.CENTER_VERTICAL
  thumbView.setLayoutParams(thumbParams)
  thumbView.setBackground(thumbDrawable)
  switchFrame.addView(thumbView)

  local function updateVisual(progress)
    if progress > 0.5 then
      trackDrawable.setColor(M3.Primary)
      trackDrawable.setStroke(0, 0)
    else
      trackDrawable.setColor(M3.SurfaceVariant)
      trackDrawable.setStroke(dp2px(2), M3.Outline)
    end

    local curSize = math.floor(thumbOffSize + (thumbOnSize - thumbOffSize) * progress + 0.5)
    thumbParams.width = curSize
    thumbParams.height = curSize

    local minLeft = marginMin
    local maxLeft = trackW - thumbOnSize - marginMax
    local curLeft = math.floor(minLeft + (maxLeft - minLeft) * progress + 0.5)
    thumbParams.leftMargin = curLeft

    if progress > 0.5 then
      thumbDrawable.setColor(M3.OnPrimary)
    else
      thumbDrawable.setColor(M3.Outline)
    end

    thumbView.setLayoutParams(thumbParams)
  end

  updateVisual(isChecked and 1.0 or 0.0)

  switchFrame.setOnClickListener(View.OnClickListener{
    onClick = function()
      isChecked = not isChecked
      local startVal = isChecked and 0.0 or 1.0
      local endVal = isChecked and 1.0 or 0.0

      local anim = ValueAnimator.ofFloat(float{startVal, endVal})
      anim.setDuration(220)
      anim.addUpdateListener(ValueAnimator.AnimatorUpdateListener{
        onAnimationUpdate = function(animation)
          local val = animation.getAnimatedValue()
          updateVisual(val)
        end
      })
      anim.start()

      if onCheckChanged then
        onCheckChanged(isChecked)
      end
    end
  })

  return switchFrame
end
